/**
 * Cloudflare Relay Worker
 *
 * Generic proxy relay — forwards requests to any target URL.
 * YouTube sees Cloudflare's IP instead of the bot's hosting IP.
 *
 * Usage:
 *   GET /api/search?q=test
 *   Headers: x-relay-target: https://www.youtube.com
 *            x-relay-path: /results?search_query=test
 *
 * Deploy: npx wrangler deploy
 */

export default {
  async fetch(request, env, ctx) {
    const target = request.headers.get("x-relay-target");
    const relayPath = request.headers.get("x-relay-path") || "/";

    if (!target) {
      return new Response(JSON.stringify({ error: "Missing x-relay-target header" }), {
        status: 400,
        headers: { "content-type": "application/json" },
      });
    }

    // Build the target URL
    const targetUrl = target.replace(/\/+$/, "") + relayPath;

    // Forward the request
    const newRequestInit = {
      method: request.method,
      headers: new Headers(request.headers),
    };

    // Forward body for non-GET requests
    if (request.method !== "GET" && request.method !== "HEAD") {
      newRequestInit.body = request.body;
      newRequestInit.duplex = "half";
    }

    // Remove relay-specific headers before forwarding
    newRequestInit.headers.delete("x-relay-target");
    newRequestInit.headers.delete("x-relay-path");
    newRequestInit.headers.delete("host");

    // Add browser-like headers
    newRequestInit.headers.set("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36");

    try {
      const response = await fetch(targetUrl, newRequestInit);

      // Return response with CORS headers
      const newHeaders = new Headers(response.headers);
      newHeaders.set("Access-Control-Allow-Origin", "*");
      newHeaders.set("Access-Control-Allow-Methods", "GET, OPTIONS");
      newHeaders.set("Access-Control-Allow-Headers", "Content-Type, X-Relay-Target, X-Relay-Path, X-Relay-Key");

      return new Response(response.body, {
        status: response.status,
        headers: newHeaders,
      });
    } catch (error) {
      return new Response(JSON.stringify({ error: error.message }), {
        status: 502,
        headers: { "content-type": "application/json" },
      });
    }
  },
};
